# Requirements for the SteadyCut pipeline
# (steadycut_pipeline.py → pipelinev3.py + shot_classifier.py + xml_assembler.py)
#
# Install with:  pip install -r requirements.txt
#
# Note: the packaged .app/.exe ships ffmpeg inside the bundle (downloaded at
# build time by CI). For development: brew install ffmpeg (macOS) or
# apt install ffmpeg (Linux/Debian). No manual step for the packaged app.

# ── Phase 1 + 2: Proxy generation & stability analysis (pipelinev3.py) ────
opencv-python-headless   # Lucas-Kanade optical flow, frame I/O
numpy                    # Array / matrix operations

# ── Phase 3: YOLO shot classification (shot_classifier.py) ────────────────
ultralytics              # YOLOv8 person detection (auto-downloads model weights)
# opencv-python-headless and numpy are shared — already listed above

# ── Phase 4: FCP7 XML assembly (xml_assembler.py) ─────────────────────────
# Uses only stdlib (xml.etree.ElementTree, xml.dom.minidom) — no extra deps.

# ── Web UI (run.py) ────────────────────────────────────────────────────────
fastapi>=0.111
uvicorn[standard]>=0.29
pywebview>=5.0      # native OS window (WKWebView/WebView2) — replaces browser launch

# ── Packaging dependencies (also needed at runtime inside the exe) ─────────
# FFmpeg is bundled into the build (see steadycut.spec + CI). The runtime
# fallback download uses stdlib zipfile — no py7zr needed.
requests>=2.31   # explicit pin (transitive dep of ultralytics, used directly)
